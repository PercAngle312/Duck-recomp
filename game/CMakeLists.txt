cmake_minimum_required(VERSION 3.23)
project(DuckDodgersGameWindowsPreflight LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
option(DUCK_BUILD_GAME "Build the private generated Duck Dodgers native game" OFF)
enable_testing()
if(WIN32)
  add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
endif()
if(NOT DUCK_BUILD_GAME)
  message(STATUS "PHASE13: portable Windows runtime preflight; full game intentionally disabled")
  foreach(component display controller paths rom_locator)
    add_executable(duck_phase13_${component} "../port/tests/${component}.cpp")
    add_test(NAME duck_phase13_${component} COMMAND duck_phase13_${component})
  endforeach()
  if(WIN32)
    add_executable(duck_phase13_mapper "../port/tests/windows_mapper.cpp")
    target_link_libraries(duck_phase13_mapper PRIVATE onecore)
    add_test(NAME duck_phase13_mapper COMMAND duck_phase13_mapper)
  endif()
  return()
endif()
set(N64RECOMP_ROOT "" CACHE PATH "Pinned N64Recomp checkout")
set(N64MODERN_ROOT "" CACHE PATH "Pinned N64ModernRuntime checkout")
set(RT64_ROOT "" CACHE PATH "Pinned RT64 checkout")
set(GENERATED_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/build" CACHE PATH "Locally generated game source (not public)")
foreach(required_path
  "${N64RECOMP_ROOT}/include/recomp.h"
  "${N64MODERN_ROOT}/CMakeLists.txt"
  "${RT64_ROOT}/CMakeLists.txt"
  "${GENERATED_ROOT}/modules/funcs_0.c"
  "${GENERATED_ROOT}/resident/funcs_0.c")
  if(NOT EXISTS "${required_path}")
    message(FATAL_ERROR "Full DuckDodgers.exe blocked: required input missing: ${required_path}. Supply local/private source tree and pinned dependencies.")
  endif()
endforeach()
message(FATAL_ERROR "Source validation passed but full game target not imported into public CI. See docs/SOURCE-IMPORT.md; do not publish translated proprietary code.")
